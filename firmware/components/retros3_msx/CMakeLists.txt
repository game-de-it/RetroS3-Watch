set(retros3_sources retros3_msx.c z80.c vdp.c media.c)

if(COMMAND idf_component_register)
    idf_component_register(SRCS ${retros3_sources} INCLUDE_DIRS "." REQUIRES heap)
    # The emulation task runs with the flash cache enabled. Cross-file
    # optimization can specialize the Z80 bus callbacks for the watch.
    target_compile_options(${COMPONENT_LIB} PRIVATE -O3 -flto -fjump-tables -ftree-switch-conversion
                           -Wall -Wextra -Werror)
    target_link_options(${COMPONENT_LIB} INTERFACE -flto)
else()
    cmake_minimum_required(VERSION 3.16)
    project(retros3_msx LANGUAGES C)
    add_library(retros3_msx STATIC ${retros3_sources})
    set_target_properties(retros3_msx PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_include_directories(retros3_msx PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_options(retros3_msx PRIVATE -Wall -Wextra -Werror)
    add_executable(retros3-msx host.c)
    target_compile_options(retros3-msx PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(retros3-msx PRIVATE retros3_msx)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(retros3-msx-test test_core.c)
        target_compile_options(retros3-msx-test PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(retros3-msx-test PRIVATE retros3_msx)
        add_test(NAME independent_msx_core COMMAND retros3-msx-test)
        set(bios_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../../sdcard/msx/bios")
        if(EXISTS "${bios_dir}/MSX2.ROM")
            add_test(NAME cbios_msx1_cartridge COMMAND retros3-msx-test 1 "${bios_dir}/MSX.ROM")
            add_test(NAME cbios_msx2_cartridge COMMAND retros3-msx-test 2 "${bios_dir}/MSX2.ROM" "${bios_dir}/MSX2EXT.ROM")
        endif()
        if(EXISTS "${bios_dir}/MSX2P.ROM")
            add_test(NAME cbios_msx2_plus_cartridge COMMAND retros3-msx-test 3 "${bios_dir}/MSX2P.ROM" "${bios_dir}/MSX2PEXT.ROM")
        endif()
    endif()
endif()
